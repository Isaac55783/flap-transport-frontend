import React, { useEffect, useRef, useState } from 'react';

import {
  StyleSheet,
  View,
  Text,
  TouchableOpacity,
  Alert,
  ActivityIndicator,
  Platform,
} from 'react-native';

import {
  Map,
  Camera,
  ViewAnnotation,
} from '@maplibre/maplibre-react-native';

import * as Location from 'expo-location';

import { useAppSelector } from '../store';

// ==================================================
// MAP STYLE
// ==================================================

const MAP_STYLE =
  'https://demotiles.maplibre.org/style.json';

// ==================================================
// TYPES
// ==================================================

type TroTro = {
  id: number;
  name: string;
  latitude: number;
  longitude: number;
  route: string;
  availableSeats: number;
};

// SAMPLE TRO-TRO DATA
// ==================================================

];

// ==================================================
// DEFAULT LOCATION
// ==================================================

const DEFAULT_LATITUDE = 5.6037;
const DEFAULT_LONGITUDE = -0.1870;

// ==================================================
// SCREEN
// ==================================================

export default function PassengerDashboard({
  navigation,
}: any) {
  // ------------------------------------------------
  // USER
  // ------------------------------------------------

  const user = useAppSelector(
    (state: any) => state.auth.user
  );

  const userName = user?.name || 'Passenger';

  // ------------------------------------------------
  // LOCATION
  // ------------------------------------------------

  const [location, setLocation] =
    useState<Location.LocationObject | null>(null);

  const [locationLoading, setLocationLoading] =
    useState(true);

  // ------------------------------------------------
  // TRO-TROS
  // ------------------------------------------------

  const [selectedTroTro, setSelectedTroTro] =
    useState<TroTro | null>(null);

  const [troTros] =
    useState<TroTro[]>(initialTroTros);

  // ------------------------------------------------
  // CAMERA
  // ------------------------------------------------

  // MapLibre v11 exposes camera methods through
  // the Camera ref.
  //
  // Using any here avoids TypeScript problems with
  // the internal CameraRef type while retaining
  // the native camera methods.

  const cameraRef = useRef<any>(null);

  // ==================================================
  // GET LOCATION
  // ==================================================

  useEffect(() => {
    let mounted = true;

    const getLocation = async () => {
      try {
        const { status } =
          await Location.requestForegroundPermissionsAsync();

        if (status !== 'granted') {
          Alert.alert(
            'Location Permission',
            'FLAP needs your location to show nearby tro-tros.'
          );

          if (mounted) {
            setLocationLoading(false);
          }

          return;
        }

        const currentLocation =
          await Location.getCurrentPositionAsync({
            accuracy: Location.Accuracy.High,
          });

        if (mounted) {
          setLocation(currentLocation);
        }
      } catch (error) {
        console.error(
          'Location error:',
          error
        );

        if (mounted) {
          Alert.alert(
            'Location Error',
            'We could not get your current location. Please try again.'
          );
        }
      } finally {
        if (mounted) {
          setLocationLoading(false);
        }
      }
    };

    getLocation();

    return () => {
      mounted = false;
    };
  }, []);

  // ==================================================
  // MAP CENTER
  // ==================================================

  const mapLatitude =
    location?.coords?.latitude ??
    DEFAULT_LATITUDE;

  const mapLongitude =
    location?.coords?.longitude ??
    DEFAULT_LONGITUDE;

  // ==================================================
  // CENTER ON USER
  // ==================================================

  const centerOnUser = () => {
    if (!location) {
      Alert.alert(
        'Location unavailable',
        'Your current location is not available yet.'
      );

      return;
    }

    cameraRef.current?.flyTo(
      [
        location.coords.longitude,
        location.coords.latitude,
      ],
      800
    );
  };

  // ==================================================
  // TRO-TRO MARKER
  // ==================================================

  const renderTroTroMarker = (
    troTro: TroTro
  ) => {
    return (
      <ViewAnnotation
        key={`trotro-${troTro.id}`}
        id={`trotro-${troTro.id}`}
        lngLat={[
          troTro.longitude,
          troTro.latitude,
        ]}
        onSelect={() => {
          setSelectedTroTro(troTro);
        }}
      >
        <View style={styles.markerContainer}>
          <View style={styles.trotroMarker}>
            <Text style={styles.markerEmoji}>
              🚌
            </Text>
          </View>
        </View>
      </ViewAnnotation>
    );
  };

  // ==================================================
  // PASSENGER LOCATION MARKER
  // ==================================================

  const renderUserMarker = () => {
    if (!location) {
      return null;
    }

    return (
      <ViewAnnotation
        id="passenger-location"
        lngLat={[
          location.coords.longitude,
          location.coords.latitude,
        ]}
      >
        <View style={styles.userMarker}>
          <View style={styles.userMarkerDot} />
        </View>
      </ViewAnnotation>
    );
  };

  // ==================================================
  // SCREEN
  // ==================================================

  return (
    <View style={styles.container}>

      {/* HEADER */}

      <View style={styles.header}>
        <View>
          <Text style={styles.welcome}>
            Welcome back
          </Text>

          <Text style={styles.name}>
            {userName}
          </Text>
        </View>

        <TouchableOpacity
          style={styles.profileButton}
          onPress={() =>
            navigation.navigate('Profile')
          }
        >
          <Text style={styles.profileText}>
            👤
          </Text>
        </TouchableOpacity>
      </View>

      {/* SEARCH */}

      <View style={styles.searchContainer}>
        <TouchableOpacity
          style={styles.searchBox}
          onPress={() =>
            navigation.navigate('Search')
          }
        >
          <Text style={styles.searchIcon}>
            🔍
          </Text>

          <Text style={styles.searchPlaceholder}>
            Where are you going?
          </Text>
        </TouchableOpacity>
      </View>

      {/* MAP */}

      <View style={styles.mapContainer}>

        {locationLoading ? (
          <View style={styles.loadingContainer}>
            <ActivityIndicator size="large" />

            <Text style={styles.loadingText}>
              Finding your location...
            </Text>
          </View>
        ) : (
          <Map
            style={styles.map}
            mapStyle={MAP_STYLE}
          >

            {/* CAMERA */}

            <Camera
              ref={cameraRef}
              zoom={12}
              initialViewState={{
                center: [
                  mapLongitude,
                  mapLatitude,
                ],
                zoom: 12,
              }}
            />

            {/* PASSENGER */}

            {renderUserMarker()}

            {/* TRO-TROS */}

            {troTros.map(
              renderTroTroMarker
            )}

          </Map>
        )}

        {/* LOCATION BUTTON */}

        <TouchableOpacity
          style={styles.locationButton}
          onPress={centerOnUser}
          activeOpacity={0.8}
        >
          <Text style={styles.locationButtonText}>
            ◎
          </Text>
        </TouchableOpacity>

      </View>

      {/* SELECTED TRO-TRO */}

      {selectedTroTro && (
        <View style={styles.trotroCard}>

          <View style={styles.cardTop}>

            <View>
              <Text style={styles.trotroName}>
                {selectedTroTro.name}
              </Text>

              <Text style={styles.routeText}>
                {selectedTroTro.route}
              </Text>
            </View>

            <TouchableOpacity
              onPress={() =>
                setSelectedTroTro(null)
              }
            >
              <Text style={styles.closeText}>
                ✕
              </Text>
            </TouchableOpacity>

          </View>

          <View style={styles.infoRow}>

            <Text style={styles.seatText}>
              🪑 {selectedTroTro.availableSeats}{' '}
              seats available
            </Text>

            <Text style={styles.liveText}>
              ● LIVE
            </Text>

          </View>

          <TouchableOpacity
            style={styles.requestButton}
            onPress={() => {
              Alert.alert(
                'Tro-Tro Selected',
                `You selected ${selectedTroTro.name}.`
              );
            }}
          >
            <Text
              style={styles.requestButtonText}
            >
              Request Ride
            </Text>
          </TouchableOpacity>

        </View>
      )}

      {/* BOTTOM ACTIONS */}

      {!selectedTroTro && (
        <View style={styles.bottomPanel}>

          <TouchableOpacity
            style={styles.actionButton}
            onPress={() =>
              navigation.navigate('MyTrips')
            }
          >
            <Text style={styles.actionIcon}>
              🚌
            </Text>

            <Text style={styles.actionText}>
              My Trips
            </Text>
          </TouchableOpacity>

          <TouchableOpacity
            style={styles.actionButton}
            onPress={() =>
              navigation.navigate('Favorites')
            }
          >
            <Text style={styles.actionIcon}>
              ⭐
            </Text>

            <Text style={styles.actionText}>
              Favorites
            </Text>
          </TouchableOpacity>

          <TouchableOpacity
            style={styles.actionButton}
            onPress={() =>
              navigation.navigate(
                'Notifications'
              )
            }
          >
            <Text style={styles.actionIcon}>
              🔔
            </Text>

            <Text style={styles.actionText}>
              Alerts
            </Text>
          </TouchableOpacity>

        </View>
      )}

    </View>
  );
}
